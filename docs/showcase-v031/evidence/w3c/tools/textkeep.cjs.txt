// Every word of BASE's WORLD entry (+ the SYSTEM-side archive note) must still be in the document after a variant
// is applied (textContent, case-insensitive: closed <details> and text-transform do not count as deletion).
const { chromium } = require('playwright'); const fs = require('fs');
const [url, ...vfs] = process.argv.slice(2);
(async () => { const b = await chromium.launch(); const words = s => new Set((s.toLowerCase().match(/[\p{L}\p{N}]+/gu) || []));
  const p0 = await b.newPage(); await p0.goto(url, { waitUntil: 'load' });
  const baseTxt = await p0.evaluate(() => document.querySelector('#world-guide').textContent + ' ' + document.querySelector('.show-lore-link').textContent);
  const W = words(baseTxt); const out = {};
  for (const vf of vfs) { const p = await b.newPage(); await p.goto(url, { waitUntil: 'load' }); await p.evaluate(fs.readFileSync(vf, 'utf8'));
    const all = words(await p.evaluate(() => document.body.textContent)); out[vf] = [...W].filter(w => !all.has(w)); }
  console.log(JSON.stringify({ baseWords: W.size, missing: out })); await b.close(); })();
