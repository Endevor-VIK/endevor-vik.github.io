def palette(p):
    im = Image.open(p).convert('RGB').resize((418, 235)); n = mag = red = 0; lum = 0.0
    for r, g, b in im.getdata():
        n += 1; h, s, v = colorsys.rgb_to_hsv(r/255, g/255, b/255); hd = h * 360
        if s > .25 and v > .2:
            mag += 260 <= hd <= 340; red += hd >= 345 or hd <= 15
        lum += .2126*r + .7152*g + .0722*b
    return {'magenta_purple_pct': round(100*mag/n, 2), 'red_pct': round(100*red/n, 1), 'mean_luminance_0_255': round(lum/n)}
