// quick look: threshold after menu jump, chapter plate I, OMEGA-0 scene (after its reveal), epilogue area; usage: node look.cjs <url> <outPrefix> <w>x<h>@<dpr>
const { chromium } = require('playwright');
const [url, out, vp] = process.argv.slice(2); const [wh, d] = vp.split('@'); const [w, h] = wh.split('x').map(Number);
(async () => { const b = await chromium.launch(); const phone = w < 700;
  const ctx = await b.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: +d, isMobile: phone, hasTouch: phone });
  await ctx.addInitScript(() => { try { sessionStorage.setItem('axiom_loader_seen', '1'); } catch (_) {} });
  const p = await ctx.newPage(); const errs = []; p.on('pageerror', e => errs.push(String(e))); p.on('console', m => { if (m.type() === 'error') errs.push(m.text()); });
  await p.goto(url, { waitUntil: 'load' }); await p.evaluate(() => document.fonts.ready);
  const settle = () => p.evaluate(() => new Promise(r => { let l = scrollY, s = 0, n = 0; const t = () => { n++; if (Math.abs(scrollY - l) < .5) s++; else s = 0; l = scrollY; if (s >= 10 || n > 500) r(); else requestAnimationFrame(t); }; requestAnimationFrame(t); }));
  const imgs = () => p.evaluate(() => Promise.all([...document.images].filter(i => { const r = i.getBoundingClientRect(); return r.bottom > 0 && r.top < innerHeight && r.width; }).map(i => i.complete ? i.decode().catch(() => {}) : new Promise(res => { i.onload = i.onerror = res; setTimeout(res, 6000); }))));
  const shot = async (n) => { await imgs(); await p.waitForTimeout(900); await p.screenshot({ path: `${out}-${n}.png` }); };
  await p.locator(w > 1080 ? '.menu a[href="#world-guide"]' : '.show-mobile-nav a[href="#world-guide"]').click(); await settle(); await shot('threshold');
  for (const [sel, name, off] of [['.ax-chapter-plate--w06', 'plate1', 0.2], ['.ax-omega', 'omega', 0.08]]) {
    await p.evaluate(([s, o]) => { document.documentElement.style.scrollBehavior = 'auto'; const e = document.querySelector(s); scrollTo(0, e.getBoundingClientRect().top + scrollY - innerHeight * o); }, [sel, off]);
    await p.waitForTimeout(2200); await shot(name);
  }
  // the return link
  await p.locator('.ax-omega-cta').click(); await p.waitForTimeout(1500);
  const sel = await p.evaluate(() => ({ id: document.getElementById('v23-map-id').textContent, focus: document.activeElement.dataset.canonId, pressed: document.querySelector('.v23-map-node[data-canon-id="02.05"]').getAttribute('aria-pressed') }));
  await shot('return');
  console.log(vp, JSON.stringify(sel), 'overflow', await p.evaluate(() => document.documentElement.scrollWidth > document.documentElement.clientWidth), 'errors', errs.length, errs.slice(0, 2));
  await b.close(); })();
