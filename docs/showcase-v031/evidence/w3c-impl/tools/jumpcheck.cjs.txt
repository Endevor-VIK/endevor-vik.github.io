const { chromium } = require('playwright');
(async () => { const b = await chromium.launch();
  for (const [w, h] of [[1024, 768], [1100, 800], [1280, 800], [1440, 900], [1600, 900], [1920, 1080], [960, 540]]) {
    const ctx = await b.newContext({ viewport: { width: w, height: h } }); await ctx.addInitScript(() => sessionStorage.setItem('axiom_loader_seen', '1'));
    const p = await ctx.newPage(); await p.goto(process.argv[2], { waitUntil: 'load' }); await p.evaluate(() => document.fonts.ready);
    await p.locator(w > 1080 ? '.menu a[href="#world-guide"]' : '.show-mobile-nav a[href="#world-guide"]').click();
    await p.evaluate(() => new Promise(r => { let l = scrollY, s = 0, n = 0; const t = () => { n++; if (Math.abs(scrollY - l) < .5) s++; else s = 0; l = scrollY; if (s >= 10 || n > 500) r(); else requestAnimationFrame(t); }; requestAnimationFrame(t); }));
    const g = await p.evaluate(() => { const c = document.querySelector('.ax-world-cta').getBoundingClientRect(); const t = document.getElementById('v22-world-title').getBoundingClientRect(); const sc = document.querySelector('.ax-world-scene').getBoundingClientRect(); return { scene: Math.round(sc.height), titleTop: Math.round(t.top), ctaBottom: Math.round(c.bottom), vh: innerHeight }; });
    console.log(w + 'x' + h, JSON.stringify(g), g.ctaBottom <= g.vh ? 'OK' : 'CUT'); await ctx.close(); }
  await b.close(); })();
